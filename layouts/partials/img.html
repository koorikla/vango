{{- /* Responsive from an asset path. Params: src (asset path), alt, class, sizes, widths (slice), loading, fetchpriority Emits AVIF and WebP srcsets with a JPEG/PNG fallback; all generated at build time. AVIF is offered first because it is materially smaller for photographs — measured on this site's own pictures at 1600px, AVIF q55 came out about 30% under WebP q80 with no difference visible at 1:1. q45 is another 35% down and does show: the stone in the sauna goes waxy and the wood loses its grain, which is the one thing these photographs are for. Not for PNG sources. Those are the room ornaments and the partner badges — flat artwork, where a lossy codec is the wrong tool and (as WebP already demonstrated on the theme art, going 109 kB to 293 kB) frequently loses to PNG outright. */ -}} {{- $src := .src -}} {{- $r := resources.Get $src -}} {{- if not $r -}} {{- warnf "img partial: asset not found: %s" $src -}} {{- else -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $loading := .loading | default "lazy" -}} {{- $widths := .widths | default (slice 480 768 1100 1600) -}} {{- $isPNG := eq (path.Ext $src) ".png" -}} {{- /* never upscale */ -}} {{- $ws := slice -}} {{- range $widths }}{{ if le . $r.Width }}{{ $ws = $ws | append . }}{{ end }}{{ end -}} {{- if not $ws }}{{ $ws = slice $r.Width }}{{ end -}} {{- $webp := slice -}} {{- range $ws }} {{- $i := $r.Resize (printf "%dx webp q80" .) -}} {{- $webp = $webp | append (printf "%s %dw" $i.RelPermalink $i.Width) -}} {{- end -}} {{- $avif := slice -}} {{- if not $isPNG }} {{- range $ws }} {{- $i := $r.Resize (printf "%dx avif q55" .) -}} {{- $avif = $avif | append (printf "%s %dw" $i.RelPermalink $i.Width) -}} {{- end -}} {{- end -}} {{- $fbSpec := cond $isPNG "png" "jpg q80" -}} {{- $fb := $r.Resize (printf "%dx %s" (index (last 1 $ws) 0) $fbSpec) -}} {{- with $avif }} {{- end }} {{- end -}}
